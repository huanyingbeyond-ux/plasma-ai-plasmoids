cmake_minimum_required(VERSION 3.16)
project(dshbridge LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)

find_package(Qt6 6.5 REQUIRED COMPONENTS Core Qml)

qt_add_qml_module(dshbridge
    URI org.dean.dsh
    VERSION 1.0
    QML_FILES
        Dummy.qml
    SOURCES
        dshbridge.cpp dshbridge.h
)

target_link_libraries(dshbridge PRIVATE Qt6::Core Qt6::Qml)
